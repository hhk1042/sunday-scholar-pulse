  schedule:
    - cron: '0 10 * * 0' # 매주 일요일 19:00 KST (10:00 UTC)

jobs:
  build-weekly-digest:
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - run: pip install requests feedparser google-genai
      - name: Fetch 19 Journals & Summarize with Gemini
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK_URL }}
        run: python scripts/fetch_and_summarize.py
      - name: Commit updated weekly JSON & deploy
        run: |
          git config user.name "sunday-scholar-bot"
          git config user.email "bot@users.noreply.github.com"
          git add data/
          git commit -m "chore: update weekly digest" || exit 0
          git push
scripts/fetch_and_summarize.py (핵심 파이프라인) 
import datetime, json, os, requests
from google import genai

# 5대 분야 19개 타겟 저널 ISSN 목록
JOURNALS_ISSN = [
  "0360-1323", "0378-7788", "0926-5805", "0003-8628", # 건축
  "0264-2751", "0042-0980", "0169-2046", "2210-6707", "0198-9715", # 도시
  "1618-8667", "0921-2973", "0301-4797", # 조경
  "1758-678X", "0959-3780", "0165-0009", "1748-9326", # 기후변화
  "2212-4209", "0921-030X", "0272-4332", "0361-3666"  # 재난재해
]

def fetch_weekly_papers():
    end_date = datetime.date.today()
    start_date = end_date - datetime.timedelta(days=7)
    issn_filter = "|".join(JOURNALS_ISSN)
    url = (
      f"https://api.openalex.org/works?filter=primary_location.source.issn:{issn_filter},"
      f"from_publication_date:{start_date},to_publication_date:{end_date}&per-page=200"
    )
    return requests.get(url, timeout=30).json().get("results", [])
